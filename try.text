#include <zephyr/kernel.h>
#include <modem/nrf_modem_lib.h>
#include <modem/lte_lc.h>
#include <nrf_modem.h>
#include <nrf_socket.h> // Required for NRF_AF_INET6 and structs

void set_manual_dns(void) {
    int err;
    
    // Trex.fi NAT64 DNS: 2001:67c:2b0::4
    struct nrf_in6_addr dns_addr = {
        .s6_addr = {0x20, 0x01, 0x06, 0x7c, 0x02, 0xb0, 0x00, 0x00,
                    0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x04}
    };

    // Use the specific helper function instead of generic configure
    // 1st arg: Family (NRF_AF_INET6)
    // 2nd arg: Pointer to the address
    err = nrf_setdnsaddr(NRF_AF_INET6, &dns_addr);
    
    if (err) {
        printk("Failed to set DNS: %d\n", err);
    } else {
        printk("Manual NAT64 DNS configured successfully!\n");
    }
}

void main(void) {
    // 1. Initialize Modem
    nrf_modem_lib_init();

    // 2. Set DNS *BEFORE* connecting
    set_manual_dns();

    // 3. Connect to network
    lte_lc_connect();
    
    // ... rest of your code
}
